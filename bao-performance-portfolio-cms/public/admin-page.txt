<script>

const $ = s => document.querySelector(s);

const MAX_IMAGE_SIZE = 5 * 1024 * 1024;

const ALLOWED_TYPES = [
  "image/jpeg",
  "image/png",
  "image/webp",
  "image/gif"
];

let activeUploads = 0;
let saving = false;
let dirty = false;

function status(el, msg, type = "") {
  el.textContent = msg;
  el.className = "status";

  if (type) {
    el.classList.add(type);
  }
}

function markDirty() {
  dirty = true;
}

function syncSaveButton() {
  $("#save").disabled = saving || activeUploads > 0;
}

function validMediaUrl(url) {
  return /^\/api\/media\/[a-zA-Z0-9_-]+\.(jpg|png|webp|gif)$/
    .test(url || "");
}

function createButton(text, className = "") {
  const button = document.createElement("button");

  button.type = "button";
  button.textContent = text;
  button.className = className;

  return button;
}

async function uploadImage(file) {

  if (!file) {
    throw new Error("Bạn chưa chọn ảnh.");
  }

  if (!ALLOWED_TYPES.includes(file.type)) {
    throw new Error(
      "Chỉ hỗ trợ JPG, PNG, WebP và GIF."
    );
  }

  if (
    file.size === 0 ||
    file.size > MAX_IMAGE_SIZE
  ) {
    throw new Error(
      "Ảnh phải có dung lượng tối đa 5 MB."
    );
  }

  const form = new FormData();

  form.append("file", file);

  const response = await fetch(
    "/api/media/upload",
    {
      method: "POST",
      body: form
    }
  );

  const data = await response.json();

  if (!response.ok) {
    throw new Error(
      data.error || "Upload ảnh thất bại."
    );
  }

  if (!validMediaUrl(data.url)) {
    throw new Error(
      "Đường dẫn ảnh trả về không hợp lệ."
    );
  }

  return data.url;
}

function createMediaPanel(project, box) {

  const panel = document.createElement("div");

  panel.className = "media-panel";

  const title = document.createElement("div");

  title.className = "media-title";
  title.textContent = "Hình ảnh đại diện";

  const preview = document.createElement("div");

  preview.className = "image-preview";

  const image = document.createElement("img");

  image.alt = project.title || "Ảnh dự án";
  image.className = "hidden";

  const placeholder = document.createElement("div");

  placeholder.className = "placeholder";

  placeholder.innerHTML =
    '<div class="placeholder-icon">＋</div>' +
    'Chưa có hình ảnh';

  preview.append(image, placeholder);

  const input = document.createElement("input");

  input.type = "file";

  input.accept =
    "image/jpeg,image/png,image/webp,image/gif";

  input.className = "hidden";

  const imageValue = document.createElement("input");

  imageValue.type = "hidden";
  imageValue.dataset.field = "image";

  imageValue.value = validMediaUrl(project.image)
    ? project.image
    : "";

  const actions = document.createElement("div");

  actions.className = "media-actions";

  const uploadButton = createButton(
    "Upload ảnh"
  );

  const removeButton = createButton(
    "Gỡ ảnh",
    "danger"
  );

  const note = document.createElement("p");

  note.className = "media-note";

  note.textContent =
    "JPG, PNG, WebP, GIF · Tối đa 5 MB. " +
    "Khuyến nghị ảnh ngang 1600 × 1000 px.";

  const mediaStatus = document.createElement("div");

  mediaStatus.className = "status hidden";

  function renderPreview() {

    if (imageValue.value) {

      image.src = imageValue.value;

      image.classList.remove("hidden");
      placeholder.classList.add("hidden");

      uploadButton.textContent = "Thay ảnh";
      removeButton.classList.remove("hidden");

    } else {

      image.removeAttribute("src");

      image.classList.add("hidden");
      placeholder.classList.remove("hidden");

      uploadButton.textContent = "Upload ảnh";
      removeButton.classList.add("hidden");
    }
  }

  image.addEventListener("error", () => {

    status(
      mediaStatus,
      "Không tải được ảnh xem trước.",
      "error"
    );

  });

  uploadButton.addEventListener("click", () => {

    if (!saving) {
      input.click();
    }

  });

  input.addEventListener("change", async () => {

    const file = input.files[0];

    if (!file || saving) {
      input.value = "";
      return;
    }

    activeUploads++;

    uploadButton.disabled = true;
    removeButton.disabled = true;

    syncSaveButton();

    status(
      mediaStatus,
      "Đang tải ảnh lên Cloudflare R2..."
    );

    try {

      const url = await uploadImage(file);

      imageValue.value = url;

      renderPreview();

      markDirty();

      status(
        mediaStatus,
        "Upload thành công. Nhấn Lưu thay đổi để cập nhật website.",
        "success"
      );

    } catch (error) {

      status(
        mediaStatus,
        error.message,
        "error"
      );

    } finally {

      activeUploads--;

      uploadButton.disabled = false;
      removeButton.disabled = false;

      input.value = "";

      syncSaveButton();
    }

  });

  removeButton.addEventListener("click", () => {

    if (saving) return;

    if (!confirm("Bạn muốn gỡ ảnh khỏi dự án này?")) {
      return;
    }

    imageValue.value = "";

    renderPreview();

    markDirty();

    status(
      mediaStatus,
      "Đã gỡ ảnh khỏi dự án. Nhấn Lưu thay đổi để xác nhận."
    );

  });

  actions.append(
    uploadButton,
    removeButton
  );

  panel.append(
    title,
    preview,
    input,
    imageValue,
    actions,
    note,
    mediaStatus
  );

  renderPreview();

  return panel;
}

function showEditor(data) {

  $("#loginPanel").classList.add("hidden");

  $("#editor").classList.remove("hidden");

  $("#logout").classList.remove("hidden");

  document
    .querySelectorAll("[data-key]")
    .forEach(el => {

      el.value = data[el.dataset.key] || "";

    });

  $("#projects").replaceChildren();

  (data.projects || []).forEach((p, i) => {

    const box = document.createElement("div");

    box.className = "project";

    const heading = document.createElement("h3");

    heading.textContent = "Mục công việc " + (i + 1);

    const grid = document.createElement("div");

    grid.className = "project-grid";

    const fields = document.createElement("div");

    for (const [field, labelText] of [

      ["title", "Tiêu đề"],

      [
        "description",
        "Mô tả / vai trò / kết quả đã xác minh"
      ]

    ]) {

      const label = document.createElement("label");

      label.textContent = labelText;

      const input = field === "title"
        ? document.createElement("input")
        : document.createElement("textarea");

      input.dataset.project = i;
      input.dataset.field = field;

      input.maxLength = field === "title"
        ? 180
        : 1000;

      input.value = p[field] || "";

      const wrap = document.createElement("div");

      wrap.className = "field";

      wrap.append(label, input);

      fields.append(wrap);

    }

    const mediaPanel = createMediaPanel(p, box);

    grid.append(fields, mediaPanel);

    box.append(heading, grid);

    $("#projects").append(box);

  });

  dirty = false;
}

async function loadEditor() {

  const response = await fetch(
    "/api/content",
    { cache: "no-store" }
  );

  if (!response.ok) {

    throw new Error(
      "Không tải được nội dung."
    );

  }

  showEditor(await response.json());
}

async function checkSession() {

  try {

    const response = await fetch(
      "/api/session",
      { cache: "no-store" }
    );

    const data = await response.json();

    if (data.authenticated) {

      await loadEditor();

    }

  } catch (error) {

    status(
      $("#loginStatus"),
      "Không kiểm tra được phiên đăng nhập.",
      "error"
    );

  }
}

// LOGIN

$("#loginForm").addEventListener(
  "submit",
  async event => {

    event.preventDefault();

    $("#loginButton").disabled = true;

    try {

      const response = await fetch(
        "/api/login",
        {
          method: "POST",
          headers: {
            "content-type": "application/json"
          },
          body: JSON.stringify({
            password: $("#password").value
          })
        }
      );

      const data = await response.json();

      if (!response.ok) {

        throw new Error(
          data.error || "Không đăng nhập được."
        );

      }

      $("#password").value = "";

      await loadEditor();

    } catch (error) {

      status(
        $("#loginStatus"),
        error.message,
        "error"
      );

    } finally {

      $("#loginButton").disabled = false;

    }

  }
);

// TRACK UNSAVED CHANGES

$("#editor").addEventListener(
  "input",
  event => {

    if (event.target.type !== "file") {
      markDirty();
    }

  }
);

// SAVE CONTENT

$("#save").addEventListener(
  "click",
  async () => {

    if (activeUploads > 0 || saving) return;

    const data = {};

    document
      .querySelectorAll("[data-key]")
      .forEach(el => {

        data[el.dataset.key] = el.value;

      });

    data.projects = [

      ...document.querySelectorAll(".project")

    ].map(box => ({

      title: box.querySelector(
        '[data-field="title"]'
      ).value,

      description: box.querySelector(
        '[data-field="description"]'
      ).value,

      image: box.querySelector(
        '[data-field="image"]'
      ).value

    }));

    saving = true;

    syncSaveButton();

    const controls = [
      ...$("#editor").querySelectorAll(
        "input, textarea, button"
      )
    ].filter(el => el.id !== "save");

    const previousStates = controls.map(
      el => el.disabled
    );

    controls.forEach(el => {
      el.disabled = true;
    });

    $("#logout").disabled = true;

    status(
      $("#saveStatus"),
      "Đang lưu nội dung và hình ảnh..."
    );

    try {

      const response = await fetch(
        "/api/content",
        {
          method: "PUT",
          headers: {
            "content-type": "application/json"
          },
          body: JSON.stringify(data)
        }
      );

      const result = await response.json();

      if (!response.ok) {

        throw new Error(
          result.error || "Lưu thất bại."
        );

      }

      dirty = false;

      status(
        $("#saveStatus"),
        "Đã lưu nội dung và hình ảnh thành công.",
        "success"
      );

    } catch (error) {

      status(
        $("#saveStatus"),
        error.message,
        "error"
      );

    } finally {

      saving = false;

      controls.forEach((el, i) => {
        el.disabled = previousStates[i];
      });

      $("#logout").disabled = false;

      syncSaveButton();

    }

  }
);

// LOGOUT

$("#logout").addEventListener(
  "click",
  async () => {

    if (saving || activeUploads > 0) {
      return;
    }

    if (
      dirty &&
      !confirm(
        "Bạn có thay đổi chưa lưu. Vẫn muốn đăng xuất?"
      )
    ) {
      return;
    }

    try {

      const response = await fetch(
        "/api/logout",
        { method: "POST" }
      );

      if (!response.ok) {
        throw new Error("Không đăng xuất được.");
      }

      dirty = false;

      location.reload();

    } catch (error) {

      status(
        $("#saveStatus"),
        error.message,
        "error"
      );

    }

  }
);

// PREVENT ACCIDENTAL EXIT

window.addEventListener(
  "beforeunload",
  event => {

    if (dirty || activeUploads > 0 || saving) {

      event.preventDefault();

      event.returnValue = "";

    }

  }
);

checkSession();

</script>
